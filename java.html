GradeService.java — Business Logic
@Service
public class GradeService {

    public double calculatePercentage(int obtained, int max) {
        if (max == 0) return 0.0;
        return ((double) obtained / max) * 100;
    }

    public String assignGrade(double pct) {
        if (pct >= 90) return "A+";
        if (pct >= 80) return "A";
        if (pct >= 70) return "B";
        if (pct >= 60) return "C";
        if (pct >= 40) return "D";
        return "F";
    }

    public StudentSummaryDTO getStudentSummary(Long studentId) {
        List<Grade> grades = gradeRepo.findByStudentId(studentId);
        int tObt = 0, tMax = 0;
        for (Grade g : grades) { tObt += g.getMarks(); tMax += g.getMax(); }
        double pct    = calculatePercentage(tObt, tMax);
        String grade  = assignGrade(pct);
        String result = pct >= 40 ? "PASS" : "FAIL";
        return new StudentSummaryDTO(tObt, tMax, pct, grade, result);
    }
}
StudentController.java — REST API
@RestController
@RequestMapping("/api/students")
public class StudentController {

    @Autowired private StudentService studentService;
    @Autowired private GradeService   gradeService;

    // GET  /api/students
    @GetMapping
    public List<Student> getAll() { return studentService.findAll(); }

    // POST /api/students
    @PostMapping
    public Student create(@RequestBody Student s) { return studentService.save(s); }

    // PUT  /api/students/{id}
    @PutMapping("/{id}")
    public Student update(@PathVariable Long id, @RequestBody Student s) {
        return studentService.update(id, s);
    }

    // DELETE /api/students/{id}
    @DeleteMapping("/{id}")
    public void delete(@PathVariable Long id) { studentService.delete(id); }

    // GET  /api/students/{id}/summary
    @GetMapping("/{id}/summary")
    public StudentSummaryDTO summary(@PathVariable Long id) {
        return gradeService.getStudentSummary(id);
    }
}
